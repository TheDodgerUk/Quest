    // ---------------------------------------------------------------------
    // High Definition Render Pipeline
    // Only spliced into the compiled .shader when com.unity.render-pipelines.high-definition
    // is present in the project. Never present otherwise, so its Core.hlsl
    // include can never fail to resolve.
    // ---------------------------------------------------------------------
    SubShader
    {
        Tags { "RenderPipeline"="HDRenderPipeline" "Queue"="Transparent" "IgnoreProjector"="True" "RenderType"="Transparent" "CanUseSpriteAtlas"="True" }
        Cull Off
        ZWrite Off
        Blend SrcAlpha OneMinusSrcAlpha

        Pass
        {
            HLSLPROGRAM
            #pragma vertex vert
            #pragma fragment frag
            // HDRP is built on the same SRP Core ShaderLibrary as URP, so the
            // shared Core.hlsl include (object->clip transforms, texture
            // macros, CBUFFER helpers) works for this simple unlit pass
            // without pulling in HDRP's full lit-shader pass library.
            #include "Packages/com.unity.render-pipelines.core/ShaderLibrary/Core.hlsl"

            struct Attributes
            {
                float4 positionOS : POSITION;
                float4 color      : COLOR;
                float2 uv         : TEXCOORD0;
            };

            struct Varyings
            {
                float4 positionHCS : SV_POSITION;
                float4 color       : COLOR;
                float2 uv          : TEXCOORD0;
            };

            TEXTURE2D(_MainTex);
            SAMPLER(sampler_MainTex);

            CBUFFER_START(UnityPerMaterial)
                float4 _Color;
                float _Radius;
                float _CurveBlend;
            CBUFFER_END

            Varyings vert(Attributes IN)
            {
                Varyings OUT;
                float safeRadius = max(_Radius, 1.0);
                float theta = IN.positionOS.x / safeRadius;
                float3 curved = float3(sin(theta) * safeRadius, IN.positionOS.y, cos(theta) * safeRadius - safeRadius);
                float3 bent = lerp(IN.positionOS.xyz, curved, _CurveBlend);
                OUT.positionHCS = TransformObjectToHClip(bent);
                OUT.uv = IN.uv;
                OUT.color = IN.color * _Color;
                return OUT;
            }

            float4 frag(Varyings IN) : SV_Target
            {
                return SAMPLE_TEXTURE2D(_MainTex, sampler_MainTex, IN.uv) * IN.color;
            }
            ENDHLSL
        }
    }
